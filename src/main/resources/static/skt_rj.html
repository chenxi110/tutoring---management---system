package com.skt.security;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import javax.crypto.SecretKey;
import java.util.Date;
import java.util.HashMap;
import java.util.Map;
@Component
public class JwtUtil {
    @Value("${jwt.secret}")
    private String secret;
    @Value("${jwt.expiration}")
    private long expiration;
    private SecretKey getKey() {
        return Keys.hmacShaKeyFor(secret.getBytes());
    }
    public String generateToken(Long id, String username, String role, String displayName) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("id", id);
        claims.put("username", username);
        claims.put("role", role);
        claims.put("displayName", displayName);
        return Jwts.builder()
                .claims(claims)
                .issuedAt(new Date())
                .expiration(new Date(System.currentTimeMillis() + expiration))
                .signWith(getKey())
                .compact();
    }
    public Claims parseToken(String token) {
        return Jwts.parser()
                .verifyWith(getKey())
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }
    public boolean validateToken(String token) {
        try {
            parseToken(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
    public Long getUserId(String token) {
        Claims c = parseToken(token);
        return c.get("id", Long.class);
    }

    public String getRole(String token) {
        return parseToken(token).get("role", String.class);
    }
    public String getUsername(String token) {
        return parseToken(token).get("username", String.class);
    }
    public String getDisplayName(String token) {
        return parseToken(token).get("displayName", String.class);
    }
}
package com.skt.security;
import io.jsonwebtoken.Claims;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.PrintWriter;
@Component
public class JwtAuthFilter implements HandlerInterceptor {
    @Autowired
    private JwtUtil jwtUtil;
    @Override
    public boolean preHandle(HttpServletRequest req, HttpServletResponse res, Object handler) throws Exception {
        String path = req.getRequestURI();
        if ("OPTIONS".equalsIgnoreCase(req.getMethod())) {
            return true;
        }
        if (isPublicPath(path)) {
            return true;
        }
        String authHeader = req.getHeader("Authorization");
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            res.setStatus(401);
            res.setContentType("application/json;charset=UTF-8");
            PrintWriter w = res.getWriter();
            w.write("{\"code\":401,\"error\":\"未登录，请先登录\"}");
            return false;
        }
        String token = authHeader.substring(7);
        if (!jwtUtil.validateToken(token)) {
            res.setStatus(401);
            res.setContentType("application/json;charset=UTF-8");
            PrintWriter w = res.getWriter();
            w.write("{\"code\":401,\"error\":\"登录已过期，请重新登录\"}");
            return false;
        }
        Claims claims = jwtUtil.parseToken(token);
        req.setAttribute("userId", claims.get("id", Long.class));
        req.setAttribute("username", claims.get("username", String.class));
        req.setAttribute("role", claims.get("role", String.class));
        req.setAttribute("displayName", claims.get("displayName", String.class));
        return true;
    }
    private boolean isPublicPath(String path) {
        if (path.startsWith("/api/auth/login") || path.startsWith("/api/auth/register")) {
            return true;

        }
        if (path.startsWith("/api/sse")) {
            return true;
        }
        if (path.startsWith("/api/share/validate")) {
            return true;
        }
        if (path.equals("/health") || path.equals("/")) {
            return true;
        }
        if (!path.startsWith("/api/")) {
            return true;
        }
        return false;
    }
}

            Long parentId = RoleAccess.getUserId(req);
            List<Map<String, Object>> children = studentService.listByParent(parentId);
            boolean isBound = children.stream().anyMatch(c -> {

            result.put("error", "学生不存在");

            result.put("code", 200);
            result.put("data", student);
        }
        return result;

    setBaseURL(url) {
        var finalUrl = (url || '/api').trim();
        if (finalUrl.indexOf('http://') > -1 || finalUrl.indexOf('https://') > -1 || finalUrl.indexOf('localhost') > -1 || finalUrl.indexOf('127.0.0.1') > -1) {
            finalUrl = '/api';
        }
        this.baseURL = '/api';
        localStorage.setItem('skt_api_base', this.baseURL);
    }
    setAuth(token, user) {
        this.token = token;
        this.user = user;
        if (typeof window !== 'undefined') {
            window.currentUser = user || null;
            window.currentRole = user && user.role ? user.role : null;
        }
        localStorage.setItem('skt_token', token);
        localStorage.setItem('skt_user', JSON.stringify(user));
        localStorage.setItem('skt_user_role', user && user.role ? user.role : '');
        sessionStorage.setItem('skt_user_role', user && user.role ? user.role : '');
    }
    clearAuth() {
        this.token = '';
        this.user = null;
        if (typeof window !== 'undefined') {
            window.currentUser = null;
            window.currentRole = null;
        }

        ['skt_token','skt_user','skt_user_role','skt_sidebar_role','skt_menu_cache','skt_sidebar_cache','skt_last_role'].forEach(function(key){

    async request(method, path, body) {
        var url = this.baseURL + path;
        var options = {
            method: method,
            headers: {
                'Content-Type': 'application/json',
                ...(this.token ? { 'Authorization': 'Bearer ' + this.token } : {})
            }
        };
        if (body && ['POST', 'PUT', 'PATCH'].includes(method)) {
            options.body = JSON.stringify(body);
        }
        let res;
        try {
            res = await fetch(url, options);
        } catch (e) {
            throw new Error('网络请求失败，请检查网络或稍后重试');
        }
        if (res.status === 401) {
            this.clearAuth();
            window.location.reload();
            throw new Error('登录已过期');
        }
        let text = null;
        try {
            text = await res.text();
        } catch (e) {
            text = null;
        }
        let data = null;
        if (text && text.length > 0) {
            try {
                data = JSON.parse(text);
            } catch (e) {
                data = { error: text };
            }
        }
        if (!res.ok) {
            const errMsg = (data && (data.error || data.msg)) || res.statusText || ('请求失败 (' + res.status + ')');
            throw new Error(errMsg);
        }
        return data === null ? {} : data;
    }

    connectSSE() {
        if (!this.token) return;
        this.closeSSE();
        var self = this;
        this._sseFailed = false;
        this.sseEventSource = new EventSource(this.baseURL + '/sse?token=' + encodeURIComponent(this.token));
        this.sseEventSource.onmessage = function(event) {
            try {
                var data = JSON.parse(event.data);
                self.messageCallbacks.forEach(function(cb) { cb(data); });
            } catch (e) {}
        };
        this.sseEventSource.onopen = function() {
            self._sseFailed = false;
            self.stopPolling();
        };
        this.sseEventSource.onerror = function() {
            self._sseFailed = true;
            self.closeSSE();
            self.startPolling();
        };
    }
    closeSSE() {
        if (this.sseEventSource) {
            this.sseEventSource.close();
            this.sseEventSource = null;
        }

    }
    startPolling() {
        if (this._pollingTimer) return;
        if (!this.token) return;
        var self = this;
        this._pollingTimer = setInterval(async function() {
            if (!self.token) { self.stopPolling(); return; }
            try {
                var res = await self.request('GET', '/messages/unread/count');
                if (res && res.code === 200 && res.count !== undefined) {
                    self.messageCallbacks.forEach(function(cb) { cb({ type: 'poll_check', count: res.count }); });
                }
            } catch (e) {}
        }, 5000);
    }
    stopPolling() {
        if (this._pollingTimer) {
            clearInterval(this._pollingTimer);
            this._pollingTimer = null;
        }
    }
    onMessage(callback) { this.messageCallbacks.push(callback); }

    _addToPendingQueue(method, path, body) {
        this.pendingQueue.push({ method: method, path: path, body: body, timestamp: Date.now() });
        localStorage.setItem('skt_pending_queue', JSON.stringify(this.pendingQueue));
    }
    async _flushPendingQueue() {
        if (this.pendingQueue.length === 0) return;
        var queue = this.pendingQueue.slice();
        this.pendingQueue = [];
        localStorage.setItem('skt_pending_queue', JSON.stringify(this.pendingQueue));
        for (var i = 0; i < queue.length; i++) {
            try {
                await this.request(queue[i].method, queue[i].path, queue[i].body);
            } catch (e) {
                this.pendingQueue.push(queue[i]);
            }
        }
        localStorage.setItem('skt_pending_queue', JSON.stringify(this.pendingQueue));
    }

function confirmPickStatus(idx, status, modal){
    const student=currentClassroomSession.students[idx];
    const rules=currentClassroomSession.scoreRules;
    const statusNames={normal:'正常应答',late:'迟到应答',absent:'缺勤',refuse:'拒答'};
    const scoreDelta=rules[status];
    student.picked=true;
    student.pickStatus=status;
    student.pickTime=new Date().toLocaleTimeString();
    student.score+=scoreDelta;
    currentClassroomSession.pickerHistory.push({

        name:student.name,
        time:student.pickTime,
        status:status,
        scoreDelta:scoreDelta
    });
    currentClassroomSession.scoreLog.push({
        student:student.name,
        delta:scoreDelta,
        source:'pick',
        time:new Date().toLocaleString(),
        remark:`点名应答-${statusNames[status]}`
    });
    if(status==='normal'||status==='late'){
        const autoSignin=document.getElementById('autoSigninOnPick');
        if(autoSignin&&autoSignin.checked){
            const alreadySigned=signinRecords.some(r=>r.name===student.name);
            if(!alreadySigned){
                signinRecords.push({name:student.name,time:new Date().toLocaleTimeString(),status:'present',isLate:status==='late'});
                if(typeof updateSigninUI==='function')updateSigninUI();
            }
        }
    }
    modal.remove();
    document.getElementById('pickerBtn').disabled=false;
    document.getElementById('pickerBtn').textContent='开始随机点名';
    renderPickerHistory();
    renderScoresGrid();
    showToast(`🎉 ${student.name} - ${statusNames[status]} (${scoreDelta>=0?'+':''}${scoreDelta}分)`);
}

function renderScoresGrid(){
    if(!currentClassroomSession)return;
    const students=currentClassroomSession.students;

    const container=document.getElementById('scoresGrid');
    requestAnimationFrame(()=>{
        const existingCards=container.querySelectorAll('.score-card');
        if(existingCards.length===students.length){
            students.forEach((s,i)=>{
                const card=existingCards[i];
                if(card){
                    const scoreEl=card.querySelector('.score');
                    if(scoreEl&&scoreEl.textContent!=s.score)scoreEl.textContent=s.score;
                    const logs=currentClassroomSession.scoreLog.filter(l=>l.student===s.name);
                    const tooltip=logs.length>0?logs.map(l=>`${l.remark}: ${l.delta>=0?'+':''}${l.delta}分 (${l.time})`).join('\n'):'暂无记录';
                    card.setAttribute('title',tooltip);
                }
            });
        }else{
            container.innerHTML=students.map((s,i)=>{
                const logs=currentClassroomSession.scoreLog.filter(l=>l.student===s.name);
                const tooltip=logs.length>0?logs.map(l=>`${l.remark}: ${l.delta>=0?'+':''}${l.delta}分 (${l.time})`).join('\n'):'暂无记录';
                return `<div class="score-card" title="${escapeHtml(tooltip)}" style="cursor:pointer;" onclick="showStudentScoreDetail('${escapeAttr(s.name)}')">

                    <div class="score">${s.score}</div>

                        <button class="score-btn plus" onclick="event.stopPropagation();changeScore(${i},1)">+</button>
                    </div>
                </div>`;
            }).join('');
        }
    });
}

function processDocx(file,status){
    mammoth.extractRawText({arrayBuffer:file}).then(r=>{
        const text=r.value.replace(/\r/g,'');
        status.textContent='Word 文件解析中...';
        const questions=parseDocxToQuestions(text);
        if(questions.length>0){
            batchImportQuizzes=questions;
            showImportPreview();
            status.textContent=`Word 文件导入成功，解析出 ${questions.length} 道题目`;
        }else{
            document.getElementById('quizQuestion').value=text;
            status.textContent='Word 文件导入成功，请手动整理格式';
        }
    }).catch(()=>status.textContent='解析失败，请尝试其他格式');
}

function saveRecord(){
    const id=document.getElementById('recordId').value;
    const date=document.getElementById('recordDate').value;
    const cn=document.getElementById('recordClassName').value;
    const sessions=parseInt(document.getElementById('recordSessions').value);
    const type=document.getElementById('recordType').value;
    const semester=document.getElementById('recordSemester').value;
    const remark=document.getElementById('recordRemark').value;
    if(!date||!cn||!sessions){showToast('请填写必填项','error');return;}
    const absentNames=[];
    document.querySelectorAll('#absentStudentsCheckboxes input:checked').forEach(cb=>{absentNames.push(cb.value);});
    const ea=document.getElementById('extraAbsentStudents').value;
    if(ea.trim()){ea.split(/[,，]/).map(s=>s.trim()).filter(s=>s).forEach(n=>{if(!absentNames.includes(n))absentNames.push(n);});}
    const course=getCourseMap()[cn]||'';
    const trialStudents=getTrialStudentsFromForm();
    let newRecordId=id?parseInt(id):Date.now();
    const makeupSrcId=document.getElementById('makeupSourceRecordId').value;
    const makeupStu=document.getElementById('makeupStudentName').value;
    const isMakeupSave=!!(makeupSrcId&&makeupStu&&!id);
    if(isMakeupSave){
        const srcIdNum=parseInt(makeupSrcId);
        const srcRec=appData.records.find(r=>r.id===srcIdNum);
        if(!srcRec){showToast('未找到原缺勤记录','error');return;}
        srcRec.absent=normalizeAbsent(srcRec.absent);
        const ai=srcRec.absent.findIndex(a=>a.name===makeupStu);
        if(ai!==-1&&srcRec.absent[ai].madeUp){
            showToast('该缺勤已登记补课，请勿重复登记。如需修改请先撤销原补课标记。','error');
            return;
        }
        const mkRecord={
            id:newRecordId,date,className:cn,class:cn,course,sessions,
            absent:normalizeAbsent(absentNames),
            type,semester,remark,trialStudents
        };
        appData.records.push(mkRecord);
        if(ai!==-1){
            srcRec.absent[ai].madeUp=true;
            srcRec.absent[ai].makeupRecordId=newRecordId;
        }else{
            srcRec.absent.push({name:makeupStu,madeUp:true,makeupRecordId:newRecordId,absentRemark:''});
        }

        saveData();
        closeRecordModal();
        showToast('补课登记成功，该缺勤记录已转入已补课名单');
        logOperation('补课登记', `为 ${makeupStu} 登记补课 (${date} ${cn})`);
        const cur=document.querySelector('.page.active').id.replace('page-','');
        if(cur==='attendance'){loadAbsentLists();}else{loadDashboard();}
        return;
    }
    const absentObj=normalizeAbsent(absentNames);
    if(id){
        const i=appData.records.findIndex(r=>r.id===parseInt(id));
        if(i!==-1){
            const oldAbsent=normalizeAbsent(appData.records[i].absent);
            absentObj.forEach(na=>{
                const old=oldAbsent.find(o=>o.name===na.name);
                if(old){na.madeUp=old.madeUp;na.makeupRecordId=old.makeupRecordId;na.absentRemark=old.absentRemark;}
            });
            appData.records[i]={id:parseInt(id),date,className:cn,class:cn,course,sessions,absent:absentObj,type,semester,remark,trialStudents};
            showToast('更新成功');
            logOperation('编辑记录', `编辑 ${date} ${cn} ${course} (${sessions}节)`);
            DataSync.syncRecord(appData.records[i],'update');
        }
    }else{
        appData.records.push({id:newRecordId,date,className:cn,class:cn,course,sessions,absent:absentObj,type,semester,remark,trialStudents});
        showToast('添加成功');
        logOperation('添加记录', `添加 ${date} ${cn} ${course} (${sessions}节)`);
        DataSync.syncRecord(appData.records[appData.records.length-1],'create');
    }
    saveData();
    closeRecordModal();
    loadRecords();
}

function doBatchTransfer(fromClass) {
    const selected = getSelectedStudents(fromClass);
    const toClass = document.getElementById('batchTransferTarget').value;
    const syncHistory = document.getElementById('batchTransferSyncHistory')?.checked || false;
    if (!toClass) { showToast('请选择目标班级', 'error'); return; }
    const fromStudents = getStudentsForClass(fromClass).slice();
    const toStudents = getStudentsForClass(toClass).slice();
    let transferred = 0;
    let skipped = [];
    selected.forEach(name => {
        if (toStudents.includes(name)) { skipped.push(name); return; }
        const idx = fromStudents.indexOf(name);

        if (idx >= 0) fromStudents.splice(idx, 1);
        toStudents.push(name);
        transferred++;
        if (syncHistory) {
            appData.records.forEach(r => {
                const rc = r.className || r.class || '';
                if (rc === fromClass) { r.className = toClass; r.class = toClass; }
            });
        }
    });
    setStudentsForClass(fromClass, fromStudents);
    setStudentsForClass(toClass, toStudents);
    _studentSelection[fromClass] = new Set();
    saveData();
    if (typeof DataSync !== 'undefined' && apiService.isLoggedIn()) {
        DataSync.syncFromServer();
    }
    document.querySelector('.modal-overlay.active')?.remove();
    loadStudents();
    let msg = `成功转班 ${transferred} 人`;
    if (skipped.length > 0) msg += `，${skipped.length} 人因重名跳过`;
    showToast(msg);
    logOperation('批量转班', `${fromClass} → ${toClass} (${transferred}人)`);
}
